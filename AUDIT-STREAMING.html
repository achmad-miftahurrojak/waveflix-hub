SEBELUM SEMUANYA
Verdict Utama

Produk ini belum menjadi platform streaming. Ia adalah katalog TMDB (proksi Go → data film/seri Indonesia) dengan player = iframe vidlink.pro pihak ketiga. Tidak ada elemen <video> milik sendiri, tidak ada HLS/DASH, tidak ada ABR, tidak ada DRM. Endpoint /api/stream itu mock yang tak pernah dipanggil frontend.

2 CRITICAL keamanan. (1) JWT secret hardcoded publik di source & tidak terset di .env → siapapun bisa memalsukan token login user mana pun. (2) Konten premium di-iframe tanpa auth & tanpa signed URL → bisa disedot gratis.

Hal yang bagus: discovery kuat (skor 8/10), detail page lengkap, watchlist + history dual-mode login/guest, semua query DB terparameterisasi (nol SQL injection), tidak ada IDOR, tidak ada stored XSS, React escapes semua string.

1. JWT secret — fail-closed (P0)

// auth.go — hapus fallback publik
// sebelum: fallback hardcoded "summer-tide-dev-secret-ganti-di-produksi"
// sesudah: wajib env, mati bila kosong → attacker tak bisa sign token tanpa secret
func jwtSecret() []byte {
    s := os.Getenv("JWT_SECRET")
    if len(s) < 32 {
        log.Fatal("[auth] JWT_SECRET wajib di-set (min 32 char, acak) — matikan agar tak jalan dgn secret default")
    }
    return []byte(s)
}

2. Token ke httpOnly cookie (P1)

// auth.go — saat login/register sukses, set cookie httpOnly alih-alih return token ke JS
func setAuthCookie(w http.ResponseWriter, token string) {
    http.SetCookie(w, &http.Cookie{
        Name:     "waveflix_session",
        Value:    token,
        Path:     "/",
        HttpOnly: true,
        Secure:   true,          // produksi HTTPS
        SameSite: http.SameSiteStrictMode, // cegah CSRF
        MaxAge:   60 * 60 * 24 * 7,
    })
}

3. Signed expiring stream URL (P0) — kerangka di endpoint /api/stream

// main.go — ganti handlerStream mock dgn HMAC-signed URL yang expire.
// (Pilih salah satu: integrasikan dgn sumber stream sungguhan, atau tutup endpoint mock.)
import (
    "crypto/hmac"
    "crypto/sha256"
    "encoding/hex"
    "fmt"
    "time"
)

func signMediaURL(media string, id int, uid int64, ttl time.Duration) string {
    exp := time.Now().Add(ttl).Unix()
    msg := fmt.Sprintf("%s/%d/%d/%d", media, id, uid, exp)
    mac := hmac.New(sha256.New, jwtSecret())
    mac.Write([]byte(msg))
    return fmt.Sprintf("https://cdn.internal/%d/%s/%d?exp=%d&sig=%s",
        id, media, uid, exp, hex.EncodeToString(mac.Sum(nil)))
}
// route dgn requireAuth; client hanya lihat URL bertanda, tak bisa generate sendiri.
// Validasi server-side: exp belum lewat + HMAC cocok (ctg. media,id,uid,exp).

4. Rate limit middleware — reusable (P0)

// middleware.go — per-IP sliding window sederhana tanpa dependensi
func rateLimit(perMin int) func(http.Handler) http.Handler {
    var mu sync.Mutex
    hits := map[string][]time.Time{}
    return func(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
            ip, _, _ := net.SplitHostPort(r.RemoteAddr)
            mu.Lock()
            now := time.Now()
            cut := now.Add(-time.Minute)
            hits[ip] = append(hits[ip], now)   // filter > 1 menit
            keep := hits[ip][:0]
            for _, t := range hits[ip] { if t.After(cut) { keep = append(keep, t) } }
            hits[ip] = keep
            mu.Unlock()
            if len(keep) > perMin {
                w.Header().Set("Retry-After", "60")
                http.Error(w, "terlalu banyak request", http.StatusTooManyRequests)
                return
            }
            next.ServeHTTP(w, r)
        })
    }
}
// main.go: mux.Handle("POST /api/auth/login", rateLimit(10)(muxLogin))

5. Confirm-before-leave + resume position (P1 — UX error prevention)

// UIProvider.tsx — simpan posisi & tanyakan sebelum keluar
"use client";
import { useEffect, useRef, useState } from "react";

const LEAVE_MSG = "Video masih diputar. Yakin mau keluar?";

export function usePlayerGuard(active: boolean) {
  const armed = useRef(false);

  useEffect(() => {
    if (!active) return;
    // tanya sebelum tab ditutup / pindah halaman internal
    const onBefore = (e: BeforeUnloadEvent) => {
      if (armed.current) { e.preventDefault(); e.returnValue = LEAVE_MSG; }
    };
    window.addEventListener("beforeunload", onBefore);
    return () => window.removeEventListener("beforeunload", onBefore);
  }, [active]);

  const arm = () => { armed.current = true; };
  const disarm = () => { armed.current = false; };
  return { arm, disarm };
}
// di InlineHeroVideo: arm() saat play aktif, disarm() saat stop().
